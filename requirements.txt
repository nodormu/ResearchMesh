# Runtime dependencies for Linux CLI Research Client for Claude.
#   pip install -r requirements.txt
#   playwright install chromium    # one-time: downloads the headless browser (pip does NOT)

anthropic>=1,<2          # the ceiling stops a fresh install drifting past a breaking major;
                         # raise it only after ruff, mypy, smoke_test.py and a live call pass
mcp[cli]>=2.0,<3         # MCP client + Streamable HTTP transport
# 2.0 is a breaking major and this code uses its API on both sides:
# `Server(on_list_tools=…, on_call_tool=…)` in mcp_server.py, `streamable_http_client`
# in mcp_client.py (headers arrive via an httpx2 client, not a `headers=` argument),
# and snake_case model fields in core/tools.py (`input_schema`, `is_error`,
# `mime_type`). None of it runs on 1.x. The `<3` matches claude-agent-sdk, so the two
# can share a venv.
prompt-toolkit>=3.0.51   # interactive CLI REPL
pydantic>=2.0            # imported directly in mcp_client.py (AnyUrl); also required by mcp/anthropic
playwright>=1.40.0       # headless browser tool (core/browser.py)

# --- local tool backings -------------------------------------------------
# Each is imported lazily, inside the tool that needs it, so a missing package
# only affects that one tool: it still gets declared to Claude and returns an
# install hint if used. Drop any tool you don't want by removing its module
# from core/local_tools.py MODULES.

# 8.9.1 is a soft floor, not a hard one: it is where KernelManager grew
# `transport_encryption` (8.9.0 shipped it but broke restart, which core/kernel.py
# uses), so it is what buys CurveZMQ-encrypted sockets instead of plaintext on
# four loopback ports. Below it the tool still runs — _start_manager() falls back
# to IPC — but the traffic is unencrypted.

# provisions them.

httpx2                   # `text_embeddings`/`vision_query` (core/text_embeddings.py,

                         # modules import it directly.
piper-tts                # `speak` — local neural TTS (core/speak.py), invoked
                         # as `python3 -m piper` (its CLI entry point). **Not**
                         # `sudo apt install piper` — that installs an unrelated
                         # GTK app for configuring gaming mice, same name by
                         # coincidence (see CLAUDE.md). Playback itself needs no
                         # pip package: it shells out to PipeWire's `paplay`.
faster-whisper           # `listen` — CPU/CTranslate2 speech-to-text
                         # (core/listen.py), called in-process
                         # (`WhisperModel(...).transcribe(...)`), no CLI entry
                         # point of its own. Capture itself goes through the
                         # system `parecord` binary (not pip); this package only
                         # does the transcription step.
mido[ports-rtmidi]>=1.3  # `midi1` — MIDI 1.0 device I/O (core/midi1.py). The
                         # `[ports-rtmidi]` extra pulls in `python-rtmidi`,
                         # mido's actual port-I/O backend — plain `mido` alone
                         # can build/parse messages and read/write .mid files
                         # but can't open a real port. Import is guarded
                         # (try/except at module load, checked once in the
                         # dispatcher): missing this only disables `midi1`,
                         # same "declares itself, returns an install hint"
                         # contract as every other tool in this section.
cffi>=1.15               # `midi1` ports (core/midi1.py, _AlsaInput and
                         # _AlsaOutput): ALSA sequencer clients opened through
                         # cffi with the system libasound.so.2, so the input
                         # queue can be 2000 events (rtmidi's is 200 and loses
                         # bursts) and SysEx of any size can be sent (rtmidi's
                         # output stops at 16,353 bytes). Without it, opening a
                         # port returns an error; building and files still work.
# `memory` needs nothing beyond the standard library (core/memory.py).

# --- system binaries, not pip ------------------------------------------------
# `document_convert` shells out to these:
#   sudo apt install libreoffice   # soffice: docx/odt/xlsx/pptx <-> pdf conversion
#   sudo apt install pandoc        # markdown -> docx/odt/pdf authoring path
#
# `computer` needs two apt packages that `pip install pyautogui` does not bring.
# Both fail confusingly: without them the tool reports pyautogui as missing, or
# capture fails, even though every package above is installed.
#   sudo apt install python3-tk    # mouseinfo imports tkinter at module level
#   sudo apt install scrot         # pyscreeze's only screenshot path on X11
#
# It also needs an X11 display. Wayland sessions ignore the XTEST input it
# synthesises, so it refuses to run on one (see core/computer.py). For a
# headless or Wayland machine:
#   sudo apt install xvfb          # xvfb-run -s '-screen 0 1280x800x24' python main.py
